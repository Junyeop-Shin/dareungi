## 목표

사람이 손대지 않아도 매시·매일·매주 데이터가 들어오고, 예측과 위험 점수가 갱신되고, 대시보드가 읽을 파일이 바뀌게 한다.
사용자 요청 6번의 "API로 받아오기"가 여기다.

## 원칙

1. **멱등(idempotent):** 같은 시간대를 두 번 받아도 결과가 같다. 받을 때마다 그 시간대를 지우고 다시 넣는다.
2. **따라잡기(catch-up):** 맥북이 잠들거나 서버가 멈춰도, 다음 실행 때 빠진 시간대를 스스로 채운다.
3. **키는 서버 안에만:** 인증키는 `.env`나 서버 비밀값에만 둔다. 브라우저·대시보드 코드에 절대 넣지 않는다. 서울 API는 8088 포트 평문 HTTP라 키가 주소에 그대로 드러나기 때문이다.
4. **대시보드는 DB가 아니라 스냅샷을 읽는다:** DuckDB 파일은 한 프로세스만 쓸 수 있다. 수집기가 DB를 쓰고, 끝나면 작은 Parquet 스냅샷을 만들어 원자적으로 교체한다.
5. **공식 파일이 정답:** 반기 파일이 공개되면 API로 모은 데이터와 대조하고, 공식 파일로 덮어쓴다.

## 3A-2. 공통 모듈 — `py/common.py`

```python
import os, time, json, datetime as dt
import requests, duckdb
from dotenv import load_dotenv                    # pip install python-dotenv

load_dotenv()                                     # .env: SEOUL_API_KEY=..., DATA_GO_KR_KEY=..., ALERT_WEBHOOK=...
SEOUL = "http://openapi.seoul.go.kr:8088"

def connect(path="data/db/ops.duckdb"):
    con = duckdb.connect(path)
    con.execute("SET memory_limit = '4GB'")
    return con

def fetch(service, start, end, *params, retries=4):
    url = f"{SEOUL}/{os.environ['SEOUL_API_KEY']}/json/{service}/{start}/{end}/" + "/".join(params)
    for i in range(retries):
        try:
            r = requests.get(url, timeout=30)
            r.raise_for_status()
            js = r.json()
            body = js.get(service) or next((v for v in js.values()     # 응답 키가 서비스 이름과
                    if isinstance(v, dict) and "RESULT" in v), None)   # 다른 API 대비
            if body is None:                                   # 데이터 없음 또는 오류
                code = js.get("RESULT", {}).get("CODE")
                if code == "INFO-200":                         # 해당 데이터 없음
                    return 0, []
                raise RuntimeError(f"{service} {params}: {js.get('RESULT')}")
            return int(body["list_total_count"]), body.get("row", [])
        except Exception:
            if i == retries - 1:
                raise
            time.sleep(2 ** i)                                 # 1, 2, 4초 뒤 재시도

def fetch_all(service, *params):
    total, rows = fetch(service, 1, 1000, *params)
    for s in range(1001, total + 1, 1000):
        _, more = fetch(service, s, min(s + 999, total), *params)
        rows += more
        time.sleep(0.2)
    return total, rows

def alert(msg):
    hook = os.environ.get("ALERT_WEBHOOK")                     # 디스코드·슬랙 웹훅 주소(선택)
    print(f"[ALERT] {msg}")
    if hook:
        requests.post(hook, json={"content": msg}, timeout=10)

def log_run(con, job, status, detail=""):
    con.execute("CREATE TABLE IF NOT EXISTS job_runs (job VARCHAR, ts TIMESTAMP, status VARCHAR, detail VARCHAR)")
    con.execute("INSERT INTO job_runs VALUES (?, now(), ?, ?)", [job, status, detail])
```

동시에 두 작업이 DB를 쓰지 않게, 작업마다 잠금 파일을 쓴다(예: `data/db/.lock`이 있으면 1분 기다렸다 재시도). 또는 모든 작업을 하나의 순차 스크립트로 묶어 매시 한 번 실행한다(가장 단순).

## 3A-3. 대여이력 수집 — `py/jobs/collect_rent.py`

```python
import datetime as dt
import pandas as pd
from common import connect, fetch_all, log_run, alert

def pending_hours(con, now, backfill_days=3):
    con.execute("CREATE TABLE IF NOT EXISTS rent_hours_done (q_hour TIMESTAMP PRIMARY KEY, n BIGINT, fetched_at TIMESTAMP)")
    last = con.execute("SELECT max(q_hour) FROM rent_hours_done").fetchone()[0]
    end = now.replace(minute=0, second=0, microsecond=0) - dt.timedelta(hours=1)     # 직전에 끝난 시간대
    start = last + dt.timedelta(hours=1) if last else end - dt.timedelta(days=backfill_days)
    hours = []
    h = start
    while h <= end:
        hours.append(h); h += dt.timedelta(hours=1)
    hours.append(end - dt.timedelta(hours=2))          # 3시간 전 시간대 재수집(늦게 반영된 건)
    return sorted(set(hours))

def table_exists(con, name):
    return con.execute("SELECT count(*) FROM information_schema.tables WHERE table_name = ?", [name]).fetchone()[0] > 0

def save_hour(con, h, rows):
    if table_exists(con, "raw_rent"):
        con.execute("DELETE FROM raw_rent WHERE q_hour = ?", [h])          # 같은 시간대는 지우고 다시
    if rows:
        df = pd.DataFrame(rows).astype(str)            # 원자료는 문자열로 보관, 변환은 뷰에서
        df["q_hour"] = h
        con.register("df", df)
        if not table_exists(con, "raw_rent"):
            con.execute("CREATE TABLE raw_rent AS SELECT * FROM df LIMIT 0")
        con.execute("INSERT INTO raw_rent BY NAME SELECT * FROM df")      # 새 필드가 생기면 여기서 오류 → 스키마 변화 감지
    con.execute("INSERT OR REPLACE INTO rent_hours_done VALUES (?, ?, now())", [h, len(rows)])

def run():
    con = connect()
    try:
        for h in pending_hours(con, dt.datetime.now()):
            _, rows = fetch_all("tbCycleRentData", h.strftime("%Y-%m-%d"), h.strftime("%H"))
            save_hour(con, h, rows)
        log_run(con, "collect_rent", "ok")
    except Exception as e:
        log_run(con, "collect_rent", "fail", str(e)); alert(f"collect_rent 실패: {e}")
        raise

if __name__ == "__main__":
    run()
```

SELECT TRY_CAST(regexp_extract(BIKE_ID, '[0-9]+') AS INTEGER) AS bike_id,
       TRY_CAST(RENT_DT AS TIMESTAMP) AS rent_dt,
       TRY_CAST(RTN_DT  AS TIMESTAMP) AS rtn_dt,
       TRY_CAST(RENT_ID AS INTEGER)   AS rent_st,        -- 단계 0에서 확인한 대여소 번호 필드로
       TRY_CAST(RTN_ID  AS INTEGER)   AS rtn_st,
       TRY_CAST(USE_MIN AS DOUBLE)    AS use_min,
       TRY_CAST(USE_DST AS DOUBLE)    AS use_m,

## 3A-5. 주별 요약 갱신과 점수 계산

### 주별 요약은 이번 주만 다시

한 번 끝난 주의 요약은 변하지 않는다. 그래서 매일 **이번 주(월요일에는 지난주도)**만 05의 `bw_usage`·`bw_fail` SQL을 `trips_live` 기준으로 다시 돌려 교체한다.

```sql
DELETE FROM bw_usage WHERE wk >= ?;          -- ? = 이번 주(월요일이면 지난주) 시작일
INSERT INTO bw_usage
SELECT ...                                   -- 05 문서 2A-9 (1)과 같은 SQL, WHERE rent_dt >= ?
```

그다음 05의 `panel` 창 함수 SQL을 주별 요약 표(수백만 행) 위에서 다시 돌린다. 몇 초면 끝난다.

- 운영 DB에는 05에서 만든 과거 주별 요약(`bw_usage`, `bw_fail`, `bike_info`)을 처음에 한 번 복사해 둔다. 그래야 '마지막 신고 이후 누적'이 끊기지 않는다

### 위험 점수 — `score_risk.py` (매주 월요일)

```python
import json, numpy as np, lightgbm as lgb

spec    = json.load(open("output/models/risk_spec.json"))
booster = lgb.Booster(model_file="output/models/" + spec["model_files"]["all"])

latest = con.execute("SELECT * FROM panel WHERE wk = (SELECT max(wk) FROM panel WHERE wk < date_trunc('week', current_date))").df()
latest["wk"] = pd.to_datetime(latest["wk"])
latest = prep(latest)                                   # 06 문서의 prep()과 같은 함수를 공용 모듈로
X = latest[spec["features"]]

latest["rate_1000km"] = np.exp(booster.predict(X, raw_score=True)) * 1000 * spec["calibration_scale"]
latest["pct"] = latest.rate_1000km.rank(pct=True)
cut = spec["grade_by_weekly_percentile"]
latest["grade"] = np.select([latest.pct >= cut["점검 권장"], latest.pct >= cut["주의"], latest.pct >= cut["보통"]],
                            ["점검 권장", "주의", "보통"], default="낮음")

contrib = booster.predict(X, pred_contrib=True)[:, :-1]          # 자전거별 SHAP
names = np.array(spec["features"])
phr = spec["reason_phrases"]
latest["reasons"] = [                                             # 위험을 '높인' 변수만, 큰 순서로 3개
    [phr[f] for f, v in sorted(zip(names, row), key=lambda t: -t[1]) if v > 0 and f in phr][:3]
    for row in contrib]
```

- 범주형 변수(`bike_type`)는 학습 때와 같은 범주 값·자료형으로 넣는다
- 점수는 **완결된 주**로만 낸다(학습과 같은 조건). 이번 주 진행 중인 정보(최근 헛대여)는 `alerts_live`로 따로 보여준다
- 체크포인트 H(06)에서 '위험 점수 미사용'으로 판정됐으면 이 작업은 끄고 건강 카드용 집계만 만든다

## 3A-6. 스냅샷 발행 — `publish.py`

```python
import os
def publish(con, table, name):
    tmp, dst = f"serving/{name}.parquet.tmp", f"serving/{name}.parquet"
    con.execute(f"COPY (SELECT * FROM {table}) TO '{tmp}' (FORMAT PARQUET)")
    os.replace(tmp, dst)                        # 대시보드가 반쯤 쓰인 파일을 읽지 않게 원자적 교체

publish(con, "bike_risk_latest", "bike_risk_latest")
publish(con, "bike_card", "bike_card")          # 시민 조회용 집계(08 문서 정의)
publish(con, "forecast_weekly", "forecast_weekly")
publish(con, "alerts_live", "alerts_live")
publish(con, "data_status", "data_status")      # 각 데이터의 최신 시각(화면의 '기준 시각')
```

## 3A-7. 모니터링과 데이터 품질

```sql
-- 시간대별 건수가 지난주 같은 시간의 30% 미만이면 경보 후보
SELECT q_hour, n, lag(n, 168) OVER (ORDER BY q_hour) AS n_lastweek
FROM rent_hours_done
QUALIFY n < 0.3 * n_lastweek AND n_lastweek > 100;
```

| 점검 | 주기 | 기준 | 대응 |
|---|---|---|---|
| 시간대별 대여 건수 급감 | 매일 | 지난주 같은 시간의 30% 미만 | 3시간 뒤 재수집, 계속되면 경보 |
| 작업 실패 | 매 실행 | `job_runs.status = 'fail'` | 웹훅 경보 |
| 작업이 안 돔 | 매일 | 마지막 성공이 26시간 이상 전 | 경보(스케줄러 점검) |
| 고장신고 0건인 날 | 매일 | 평소 하루 수백 건 | API 지연인지 확인 |
| 스키마 변화 | 매 수집 | 필드 이름·개수 변경 | 수집 중단 후 매핑 수정 |

- 2026-09-25에 대여이력 API가 몇 시간 동안 평소의 일부만 돌려준 사례가 외부에 보고됐다. **건수 급감 감지와 재수집은 선택이 아니라 필수**다

### GitHub Actions를 쓸 때 (선택)

```yaml
# .github/workflows/collect.yml

  schedule: [{ cron: "15 * * * *" }]      # UTC 기준이다. 한국 시간 = UTC + 9

      - uses: actions/checkout@v4
      - uses: actions/setup-python@v5
        with: { python-version: "3.11" }
      - run: pip install -r requirements.txt
      - run: python py/jobs/collect_rent.py
        env:
          SEOUL_API_KEY: ${{ secrets.SEOUL_API_KEY }}
```

## 흔한 함정

- **대시보드가 DuckDB 파일을 직접 연다** → 수집기와 충돌한다. 스냅샷만 읽는다
- **실패하면 그냥 넘어간다** → 빈 시간대가 조용히 쌓인다. `rent_hours_done`과 건수 점검으로 잡는다
- **API 응답 필드를 숫자로 바로 바꿔 저장한다** → 형식이 바뀌면 조용히 NULL이 된다. 원자료는 문자열로 두고 뷰에서 바꾼다
- **cron 시간대를 헷갈린다** → 서버 시간대와 GitHub Actions(UTC)를 확인한다
- **점수를 진행 중인 주로 낸다** → 학습 조건과 달라진다. 완결된 주로만
