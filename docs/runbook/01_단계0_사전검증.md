## 0-1. 서울 열린데이터광장 API 호출 함수

`py/seoul_api.py`로 저장한다. 이후 모든 단계에서 불러 쓴다.

```python
import os, time
from pathlib import Path
import requests
import pandas as pd
from dotenv import load_dotenv

# 이 파일(py/seoul_api.py) 기준으로 한 단계 위 = 프로젝트 폴더의 .env를 읽는다
# → 노트북을 어느 폴더에서 실행하든 키를 찾는다
load_dotenv(Path(__file__).resolve().parent.parent / ".env")
SEOUL = "http://openapi.seoul.go.kr:8088"

def _body(js, service):
    """응답에서 데이터 본문을 찾는다.
    서비스 이름과 응답 키가 다른 API가 있다(예: bikeList → rentBikeStatus)"""
    if service in js:
        return js[service]
    return next((v for v in js.values() if isinstance(v, dict) and "RESULT" in v), None)

def seoul_get(service, start=1, end=1000, *params):
    """한 페이지(최대 1,000건) 호출. 반환: total, code, msg, rows"""
    url = f"{SEOUL}/{os.environ['SEOUL_API_KEY']}/json/{service}/{start}/{end}/" + "/".join(params)
    js = requests.get(url, timeout=30).json()
    body = _body(js, service)
    if body is None:                            # 데이터 없음·오류면 본문 없이 RESULT만 온다
        r = js.get("RESULT", js)
        return {"total": 0, "code": r.get("CODE"), "msg": r.get("MESSAGE"), "rows": [], "raw": js}
    res = body.get("RESULT", {})
    return {"total": int(body.get("list_total_count", 0)), "code": res.get("CODE"),
            "msg": res.get("MESSAGE"), "rows": body.get("row", [])}

def seoul_all(service, *params):
    """전체 페이지를 모아 DataFrame으로"""
    first = seoul_get(service, 1, 1000, *params)
    rows = list(first["rows"])
    for s in range(1001, first["total"] + 1, 1000):
        rows += seoul_get(service, s, min(s + 999, first["total"]), *params)["rows"]
        time.sleep(0.2)
    return pd.DataFrame(rows)
```

**확인된 동작 (2026-09-29 실행):** 날짜 파라미터는 '그날 하루'가 아니라 **그날부터 약 30일치**를 돌려준다(`20260701` → regDttm 07-01 00:19 ~ 07-31 23:48, 8,346행). 그래서 위 표의 `total`은 하루 건수가 아니라 30일(또는 오늘까지) 누적이다. 어제 날짜에도 값이 나오므로 **시차는 0~1일**이다.

- 응답 컬럼: `bikeNo`(예: SPB-67517), `regDttm`(초 단위 일시), `mlangComCdName`(고장구분). `START_INDEX`·`END_INDEX`·`RNUM`은 버린다
- **한 행 = 신고 1건 × 고장유형 1개.** 한 번 신고에 타이어·체인을 함께 고르면 같은 `bikeNo`·`regDttm`으로 두 행이 생긴다 → '신고 건수'(고유 bikeNo+regDttm)와 '유형 건수'(행 수)를 구분해 센다
- 하루치가 필요하면 한 번 받아서 `regDttm` 날짜로 거른다. 날짜마다 따로 부르면 같은 신고가 최대 30번 겹친다

한 달치를 받아 일별로 센다.


- [x] `types_per_report` 평균 1.14 (2026년 7월: 행 269.2 / 신고 236.2 per day, 일별 1.1~1.2로 안정) → 행으로 세면 신고 건수가 약 14% 부풀려진다

### 인코딩 확인과 UTF-8 변환

서울 데이터 CSV는 해에 따라 CP949(한글 윈도우 인코딩)와 UTF-8이 섞여 있다. pandas는 둘 다 읽을 수 있지만, DuckDB는 UTF-8만 읽으므로 변환본을 만들어 둔다.

```python
import codecs
from pathlib import Path
ROOT = Path("..")          # 노트북이 notebooks/에서 실행되므로 한 단계 위 = 프로젝트 폴더

def sniff(path, n=200_000):
    """파일 앞부분으로 형식·인코딩을 추정한다"""
    with open(path, "rb") as fh:
        raw = fh.read(n)
    if raw[:2] == b"PK":
        return "zip/xlsx"                          # 확장자만 .csv인 압축·엑셀 파일
    if raw[:2] in (b"\xff\xfe", b"\xfe\xff"):
        return "utf-16"
    if raw.lstrip()[:1] == b"<":
        return "html"                              # 다운로드가 실패해 오류 페이지가 저장된 경우
    for enc in ("utf-8-sig", "cp949"):
        try:
            codecs.getincrementaldecoder(enc)().decode(raw, final=False)
            return enc
        except UnicodeDecodeError:
            pass
    return "mixed"                                 # 대부분은 맞는데 깨진 바이트가 섞여 있다

def _best_guess(path, n=5_000_000):
    """깨진 바이트가 섞인 파일: utf-8과 cp949 중 대체 글자가 적은 쪽을 고른다"""
    with open(path, "rb") as fh:
        raw = fh.read(n)
    return min(("utf-8-sig", "cp949"), key=lambda e: raw.decode(e, errors="replace").count("�"))

def _copy(src, dst, enc, errors):
    n_bad = 0
    with open(src, encoding=enc, errors=errors, newline="") as fi, \
         open(dst, "w", encoding="utf-8", newline="") as fo:
        for line in fi:
            if errors == "replace":
                n_bad += line.count("�")
            fo.write(line)
    return n_bad

def to_utf8(src, dst):
    """원본은 그대로 두고 UTF-8 사본을 만든다. 반환: (인코딩, 깨진 글자 수)"""
    enc = sniff(src)
    if enc in ("zip/xlsx", "html"):
        return enc + " → 변환 안 함", None
    Path(dst).parent.mkdir(parents=True, exist_ok=True)
    if enc != "mixed":
        try:
            return enc, _copy(src, dst, enc, "strict")
        except UnicodeDecodeError:                 # 앞부분은 멀쩡하고 뒤에서 깨진 경우
            pass
    enc = _best_guess(src)
    return enc + " (깨진 글자는 �로 대체)", _copy(src, dst, enc, "replace")

rows = []
for f in sorted(p for p in (ROOT / "data/raw/failure").iterdir() if p.suffix.lower() == ".csv"):
    try:
        enc, n_bad = to_utf8(f, ROOT / "data/interim/failure_utf8" / f.name)
    except Exception as e:                         # 한 파일이 실패해도 나머지는 계속
        enc, n_bad = f"오류: {e}", None
    rows.append({"file": f.name, "MB": round(f.stat().st_size / 1e6, 1), "encoding": enc, "깨진 글자": n_bad})
conv_log = pd.DataFrame(rows)
conv_log
```

- 인코딩은 파일 앞부분으로 추정하고, 중간에 깨진 바이트가 있으면 utf-8·cp949 중 덜 깨지는 쪽으로 읽어 깨진 글자만 `�`로 바꾼다. **깨진 글자 수**를 기록한다(수십 개 이하면 대여소명 같은 문자열 일부라 분석에 영향 없음)
- `zip/xlsx → 변환 안 함`: 확장자만 .csv인 압축·엑셀 파일. 아래 셀로 처리한다(2026-10-01 실행: 13개 중 12개 cp949·깨진 글자 0, `22.07-12` 1개만 이름만 .csv인 엑셀 → 시트 1개, 105,550행 × 3열로 변환)

```python

```

**대여이력은 ZIP을 풀지 않고 안의 파일 앞부분만 읽어 연도별로 비교한다.** 전 기간을 받아 두었다면 2015~2026 전부를 한 번에 볼 수 있다(몇 초). 연도 ZIP을 한꺼번에 풀면 수십 GB가 되므로 풀지 않는다.

```python
import codecs, zipfile

def _ok(raw, enc):
    try:
        codecs.getincrementaldecoder(enc)().decode(raw, final=False)
        return True
    except UnicodeDecodeError:
        return False

def _head(raw):
    enc = next((e for e in ("utf-8-sig", "cp949") if _ok(raw, e)), "?")
    lines = raw.decode("cp949" if enc == "?" else enc, errors="replace").splitlines()
    return enc, (lines[0] if lines else ""), (lines[1] if len(lines) > 1 else "")

def peek_rent(folder):
    """ZIP은 풀지 않고 안의 파일 앞부분만 읽는다"""
    rows = []
    for p in sorted(Path(folder).iterdir()):
        if p.suffix.lower() == ".zip":
            with zipfile.ZipFile(p) as zf:
                for m in zf.infolist():
                    if m.is_dir():
                        continue
                    name = m.filename
                    if not m.flag_bits & 0x800:            # 윈도우에서 만든 ZIP은 한글 이름이 cp949
                        try: name = name.encode("cp437").decode("cp949")
                        except (UnicodeEncodeError, UnicodeDecodeError): pass
                    if not name.lower().endswith(".csv"):
                        rows.append({"source": p.name, "file": name, "MB": round(m.file_size / 1e6), "enc": "csv 아님"})
                        continue
                    with zf.open(m) as fh:
                        enc, header, row1 = _head(fh.read(20_000))
                    rows.append({"source": p.name, "file": name, "MB": round(m.file_size / 1e6),
                                 "enc": enc, "n_col": header.count(",") + 1, "header": header, "row1": row1})
        elif p.suffix.lower() == ".csv":
            with open(p, "rb") as fh:
                enc, header, row1 = _head(fh.read(20_000))
            rows.append({"source": p.name, "file": p.name, "MB": round(p.stat().st_size / 1e6),
                         "enc": enc, "n_col": header.count(",") + 1, "header": header, "row1": row1})
    return pd.DataFrame(rows)

rent_peek = peek_rent(ROOT / "data/raw/rent_history")
print(rent_peek.groupby("source")["MB"].sum())                 # 압축 해제 시 용량(MB) → 2A 디스크 계획
rent_peek[["source", "file", "MB", "enc", "n_col"]]
```

## 0-6. Q3 — 자전거번호 조인 매칭률 (가장 중요)

대여이력 한 달 파일은 수백만 행이라 DuckDB로 처리한다. 파이썬 안에서 쓰는 가벼운 데이터베이스로, 큰 CSV도 메모리에 다 올리지 않고 SQL로 계산한다.

```python
import duckdb
(ROOT / "data/db").mkdir(parents=True, exist_ok=True)
con = duckdb.connect(str(ROOT / "data/db/phase0.duckdb"))

con.execute(f"""CREATE OR REPLACE TABLE rent AS
  SELECT * FROM read_csv('{RENT_202601.resolve().as_posix()}', header = true, all_varchar = true)""")
con.execute(f"""CREATE OR REPLACE TABLE fail AS
  SELECT * FROM read_csv('{FAIL_2026H1.resolve().as_posix()}', header = true, all_varchar = true)""")

con.sql("DESCRIBE rent").show()        # 실제 컬럼 이름 확인
con.sql("DESCRIBE fail").show()
```

번호에서 숫자만 뽑아 정수로 맞춘 뒤 비교한다. 아래 SQL의 한글 컬럼 이름은 실제 이름으로 바꾼다.
등록일시가 `YYYY-MM-DD` 형식이 아니면(예: `20260115`) 비교하는 문자열도 같은 형식으로 바꾼다.

```python
con.sql("""
WITH r AS (
  SELECT DISTINCT CAST(regexp_extract("자전거번호", '[0-9]+') AS INTEGER) AS bike FROM rent
),
f AS (
  SELECT CAST(regexp_extract("자전거번호", '[0-9]+') AS INTEGER) AS bike FROM fail
  WHERE "등록일시" >= '2026-01-15' AND "등록일시" < '2026-02-01'
)
SELECT count(*) AS n_reports,
       avg(CASE WHEN r.bike IS NOT NULL THEN 1.0 ELSE 0 END) AS match_rate
FROM f LEFT JOIN r USING (bike)
""").show()
```

왜 1월 15~31일 신고만 보나: 1월 대여 파일 하나로 시험하므로, 신고 직전 2주 이상의 대여 기록이 있는 신고만 비교하는 것이다. 이용되지 않던 자전거가 신고될 수도 있어 매칭률은 실제보다 조금 낮게 나온다.

반대 방향과, 매칭 안 된 번호도 본다.

```python
con.sql("""
WITH r AS (SELECT DISTINCT CAST(regexp_extract("자전거번호", '[0-9]+') AS INTEGER) AS bike FROM rent),
     f AS (SELECT DISTINCT CAST(regexp_extract("자전거번호", '[0-9]+') AS INTEGER) AS bike, "자전거번호" AS raw
           FROM fail WHERE "등록일시" >= '2026-01-01' AND "등록일시" < '2026-02-01')
SELECT (SELECT count(*) FROM r) AS bikes_rented,
       (SELECT count(*) FROM f) AS bikes_reported,
       (SELECT count(*) FROM r JOIN f USING (bike)) AS both_
""").show()

con.sql("""
SELECT f."자전거번호" FROM fail f
LEFT JOIN (SELECT DISTINCT CAST(regexp_extract("자전거번호", '[0-9]+') AS INTEGER) AS bike FROM rent) r
  ON CAST(regexp_extract(f."자전거번호", '[0-9]+') AS INTEGER) = r.bike
WHERE r.bike IS NULL AND f."등록일시" >= '2026-01-15' AND f."등록일시" < '2026-02-01'
LIMIT 20
""").show()
```

**감 잡는 숫자:** 신고는 자전거 한 대당 연 약 2.5건이다. 한 달에 1건 이상 신고되는 자전거는 운행 대수의 약 15~20%가 정상 범위다.

**결과 (2026-10-01):** 1월 15~31일 신고 3,076행 중 **99.87%가 1월 대여이력과 매칭** → 단계 2 진행. 반대 방향: 1월에 대여된 자전거 36,290대, 1월에 신고된 자전거 4,461대, 둘 다 4,452대(신고 자전거의 99.8%). 매칭 안 된 4건(SPB-34468, SPB-64292, SPB-53622, SPB-63582)은 번호 형식이 같다 → 형식 문제가 아니라 1월에 대여되지 않은 자전거(정비·보관 중이거나 대여 없이 신고). 1월에 신고된 자전거 비율 12.3%는 겨울이라 위 기준(15~20%)보다 낮다.

확인
- [x] 매칭률을 기록한다
- [ ] 매칭 안 된 번호 20개를 눈으로 본다. 접두사가 다른지, 자릿수가 다른지, 새싹따릉이 같은 다른 차종인지
- [ ] 같은 숫자가 다른 차종에 겹치지 않는지 확인한다(자전거구분별 번호 범위)

판정

| 매칭률 | 판정 | 다음 행동 |
|---|---|---|
| 90% 이상 | 단계 2 진행 | 없음 |
| 70~90% | 보류 | 번호 정규화를 고쳐 다시 잰다(앞자리 0, 접두사, 차종 구분) |
| 70% 미만 | 단계 2 재설계 | 매칭되는 기간·차종으로 범위를 줄이거나, 단계 1과 집계 대시보드만 진행 |

## 0-7. Q5 — 이용거리 품질

```python
con.sql("""
SELECT count(*) AS n,
  avg(CASE WHEN TRY_CAST("이용거리(M)" AS DOUBLE) IS NULL THEN 1.0 ELSE 0 END) AS na_rate,
  avg(CASE WHEN TRY_CAST("이용거리(M)" AS DOUBLE) = 0 THEN 1.0 ELSE 0 END)     AS zero_rate,
  quantile_cont(TRY_CAST("이용거리(M)" AS DOUBLE), 0.5)  AS median_m,
  quantile_cont(TRY_CAST("이용거리(M)" AS DOUBLE), 0.99) AS p99_m
FROM rent
""").show()
```

**결과 (2026-01, 2026-10-01 실행):** 1,640,582건, 결측 0%, 거리 0 3.5%, 중앙값 1,142m, 99% 값 10.4km.

속도로도 검산하고, 거리 0이 어떤 대여인지 나눠 본다. 같은 대여소에 바로 반납한 0은 헛대여(실제 현상), **다른 대여소로 갔는데 0**이면 거리 기록 누락(데이터 오류)이다.

```python
con.sql("""
WITH t AS (
  SELECT TRY_CAST("이용거리(M)" AS DOUBLE) / 1000 AS km,
         TRY_CAST("이용시간(분)" AS DOUBLE)      AS mins,
         "대여 대여소번호" = "반납대여소번호"      AS same_st
  FROM rent
)
SELECT
  quantile_cont(km / (mins / 60), 0.5)  FILTER (WHERE mins >= 1 AND km > 0) AS speed_median,
  quantile_cont(km / (mins / 60), 0.99) FILTER (WHERE mins >= 1 AND km > 0) AS speed_p99,
  avg(CASE WHEN mins >= 1 AND km / (mins / 60) > 35 THEN 1.0 ELSE 0 END)    AS over35_rate,
  avg(CASE WHEN km = 0 AND same_st     THEN 1.0 ELSE 0 END)                  AS zero_same_st,
  avg(CASE WHEN km = 0 AND NOT same_st THEN 1.0 ELSE 0 END)                  AS zero_diff_st
FROM t
""").show()
```

- [x] 거리 0 비율과 결측 비율을 기록한다 → 결측 0%, 0은 3.5%
- [x] 속도 결과: 중앙값 8.7km/h, 99% 값 19.2km/h, 35km/h 초과 0.06%. 거리 0 중 같은 대여소 반납 2.47%(헛대여 등 실제 현상), **다른 대여소인데 0 = 0.97%(기록 누락)**
- 판정: 노출 = 이동거리 사용. 단 '다른 대여소인데 0'과 '35km/h 초과'는 05에서 `dist_bad`로 표시하고, 그 행의 거리는 대여소 간 직선거리 × 보정계수로 채운다(이용시간 × 중앙 속도는 보조)
- 참고: 속도는 데이터에 있는 열이 아니라 이용거리 ÷ 이용시간으로 계산한 **평균 속도**다(신호 대기·정차 포함)
- [ ] 속도 35km/h를 넘는 비율을 기록한다. 물리적으로 드문 값이라 데이터 오류 신호다
- [ ] 속도 중앙값이 대략 한 자릿수 후반~10km/h대면 정상일 가능성이 높다

판정
- 0·결측 비율 5% 미만 → 노출 = 이동거리를 쓸 수 있다. 단 `zero_diff_st`(다른 대여소인데 0)가 1%를 넘으면 그 행은 거리 대신 이용시간으로 보정하는 규칙을 05에 넣는다
- 높다 → 노출 = 이용시간 또는 대여 건수 (03·06 문서의 offset을 이것으로 바꾼다)

```python
# (선택) 2021-06-15 08시: 파일에는 몇 건인가 — ZIP을 풀지 않고 한 열만 읽는다(1~2분)
import io, zipfile
z = ROOT / "data/raw/rent_history/서울특별시 공공자전거 대여이력 정보_2021.zip"
with zipfile.ZipFile(z) as zf:
    def nm(m): return m.filename if m.flag_bits & 0x800 else m.filename.encode("cp437").decode("cp949")
    m = next(m for m in zf.infolist() if nm(m).endswith("2106.csv"))
    n = 0
    with zf.open(m) as fh:
        txt = io.TextIOWrapper(fh, encoding="cp949", errors="replace")
        for ch in pd.read_csv(txt, usecols=["대여일시"], dtype=str, chunksize=500_000):
            n += ch["대여일시"].str.startswith("2021-06-15 08").sum()
print("파일 2021-06-15 08시:", n, "/ API: 2,538")
```

- 파일 ≈ 2,538 → API 정상, 그날이 실제로 적었다. 파일이 훨씬 많다 → 과거 API 일부 누락(과거는 파일만 쓰는 현재 계획 유지)

## 흔한 함정

- **인증키를 코드에 써서 깃에 올린다** → `.env`에만 둔다. 올렸다면 즉시 재발급한다

- **샘플 키(`sample`)로 테스트한다** → 5건만 온다. 반드시 본인 키로
- **API 결과만 보고 파일과 비교하지 않는다** → 같은 날짜 건수 비교(0-2, 0-3)를 꼭 한다
